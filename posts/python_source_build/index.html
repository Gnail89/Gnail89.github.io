案例 - 1#

软件版本#

  • openEuler 22.03
  • openssl-1.1.1w.tar.gz
  • Python-3.9.22.tgz

openssl#

dnf install -y gcc gcc-c++ make perl perl-devel

tar zxf openssl-1.1.1w.tar.gz
./config --prefix=/opt/openssl
make && make install

Python#

dnf install -y bzip2-devel libffi-devel zlib-devel xz-devel ncurses-devel sqlite-devel readline-devel tk-devel gdbm-devel libnsl2-devel uuid-devel

./configure --prefix=/opt/python-3.9.22 --with-openssl=/opt/openssl --enable-shared --enable-optimizations
make && make install

案例 - 2#

软件版本#

  • openEuler 24.03 (LTS-SP3)
  • openssl-1.0.2u.tar.gz
  • Python-2.7.18.tgz

openssl#

# 依赖安装
dnf install -y gcc gcc-c++ make perl perl-devel

# 编译安装
tar zxf openssl-1.0.2u.tar.gz
./config --prefix=/opt/openssl-1.0.2 shared
make && make install_sw

# 配置文件
cat > /etc/ld.so.conf.d/openssl-1.0.2.conf <<EOF
/opt/openssl-1.0.2/lib
EOF

# 生效
ldconfig

python#

# 依赖安装
dnf install -y gcc make zlib-devel bzip2 bzip2-devel readline-devel sqlite sqlite-devel openssl-devel tk-devel libffi-devel wget tar gdbm-devel libdb-devel libnsl2-devel openssl-devel gdbm-devel libdb-devel

# 设置环境变量
export LDFLAGS="-L/opt/openssl-1.0.2/lib"
export CPPFLAGS="-I/opt/openssl-1.0.2/include"
export PKG_CONFIG_PATH=/opt/openssl-1.0.2/lib/pkgconfig

# 编译安装
tar zxf Python-2.7.18.tgz
./configure --prefix=/opt/python2.7 --enable-optimizations
make && make install

# 安装pip
/opt/python2.7/bin/python2 -m ensurepip --upgrade

# 验证ssl
python -c "import ssl; print(ssl.OPENSSL_VERSION)"
