使用Dockerfile构建Zabbix Agent程序#

  • 创建Dockerfile文件
# 阶段 1: 构建环境 (Builder)
FROM alpine:latest AS builder

# 1. 设置镜像源并安装编译依赖
RUN sed -i 's/dl-cdn.alpinelinux.org/mirrors.ustc.edu.cn/g' /etc/apk/repositories && \
    apk add --no-cache \
        make automake autoconf gcc g++ libc-dev pkgconfig \
        tar linux-headers \
        pcre2-dev pcre2-static

# 2. 复制源码包
COPY ./openssl-3.5.6.tar.gz /tmp/
COPY ./zabbix-7.0.25.tar.gz /tmp/

WORKDIR /tmp

# 3. 编译 OpenSSL (静态库)
RUN tar -zxvf openssl-3.5.6.tar.gz && \
    cd openssl-3.5.6 && \
    ./config --prefix=/opt/openssl no-shared no-zlib -static && \
    make -j$(nproc) && \
    make install_sw && \
    cd /tmp && rm -rf openssl-3.5.6 openssl-3.5.6.tar.gz

# 4. 编译 Zabbix Agent (静态链接)
RUN tar -zxvf zabbix-7.0.25.tar.gz && \
    cd zabbix-7.0.25 && \
    ./configure \
        --prefix=/opt/zabbix \
        --enable-agent \
        --enable-static \
        --with-libpcre2 \
        --enable-ipv6 \
        --with-openssl=/opt/openssl \
        CFLAGS="-static -std=gnu99" \
        LDFLAGS="-static" && \
    make -j$(nproc) && \
    make install && \
    cd /tmp && rm -rf zabbix-7.0.25 zabbix-7.0.25.tar.gz

# 阶段 2: 运行环境 (Output)
FROM alpine:latest AS output

# 1. 设置镜像源 (用于安装验证工具)
RUN sed -i 's/dl-cdn.alpinelinux.org/mirrors.ustc.edu.cn/g' /etc/apk/repositories

# 2. 从 builder 阶段复制编译好的二进制文件
COPY --from=builder /opt/zabbix /opt/zabbix

# 3. 安装 file 工具用于验证静态链接状态
RUN apk add --no-cache file && \
    echo "=== 文件类型验证 ===" && \
    file /opt/zabbix/sbin/zabbix_agentd && \
    echo -e "\n=== 动态链接检查 (显示: not a dynamic executable) ===" && \
    ldd /opt/zabbix/sbin/zabbix_agentd || true

# 4. 设置默认启动命令
CMD ["/opt/zabbix/sbin/zabbix_agentd", "--version"]
