  • 新增 diskimage_builder/elements/openeuler-minimal/root.d/10-openeuler-image
# cat /usr/local/lib/python3.11/site-packages/diskimage_builder/elements/openeuler-minimal/root.d/10-openeuler-image
# 参考centos元素中的配置

#!/bin/bash

if [ ${DIB_DEBUG_TRACE:-0} -gt 0 ]; then
    set -x
fi
set -eu
set -o pipefail

DIB_LOCAL_IMAGE=${DIB_LOCAL_IMAGE:-}

if [ -n "$DIB_LOCAL_IMAGE" ]; then
    IMAGE_LOCATION=$DIB_LOCAL_IMAGE
    # No need to copy a local image into the cache directory, so just specify
    # the cached path as the original path.
    CACHED_IMAGE=$IMAGE_LOCATION
    BASE_IMAGE_FILE=$(basename $DIB_LOCAL_IMAGE)
    BASE_IMAGE_TAR=$BASE_IMAGE_FILE.tgz
fi

echo "Extracting image..."
$TMP_HOOKS_PATH/bin/extract-image $BASE_IMAGE_FILE $BASE_IMAGE_TAR $IMAGE_LOCATION $CACHED_IMAGE

安装DIB和IPA builder#

pip3 install diskimage-builder

构建镜像#

  • 配置环境变量
# 指定元素路径
export ELEMENTS_PATH=/usr/local/share/ironic-python-agent-builder/dib

# 指定目标镜像中的内核和ramdisk
export DIB_BAREMETAL_INITRD_PATTERN="initramfs-6.6.0-144.0.0.130.oe2403sp3.x86_64.img"
export DIB_BAREMETAL_KERNEL_PATTERN="vmlinuz-6.6.0-144.0.0.130.oe2403sp3.x86_64"

# 开启debug信息
export DIB_DEBUG_TRACE="1"

# 指定python版本
# export DIB_PYTHON_VERSION=2

# 配置用户信息
export DIB_DEV_USER_PASSWORD="openeuler"
export DIB_DEV_USER_PWDLESS_SUDO="YES"
export DIB_DEV_USER_USERNAME="openeuler"

# 指定镜像源
export DIB_DISTRIBUTION_MIRROR="https://mirror.lzu.edu.cn/openeuler"

# 指定本地基础镜像
export DIB_LOCAL_IMAGE="/root/openEuler-24.03-LTS-SP3-x86_64.qcow2"

# 配置release信息
export DIB_RELEASE="24.03-LTS-SP3"

# 配置本地源码路径
# export DIB_REPOLOCATION_ironic_lib="/root/ironic-lib-src"
export DIB_REPOLOCATION_ironic_python_agent="/root/ironic-python-agent-src"
export DIB_REPOLOCATION_requirements="/root/requirements-src"

# 配置源码分支信息
# export DIB_REPOREF_ironic_lib="stable/train"
export DIB_REPOREF_ironic_python_agent="stable/2025.2"
export DIB_REPOREF_requirements="stable/2025.2"
  • 开始构建镜像
disk-image-create -a amd64 ironic-python-agent-ramdisk openeuler-minimal dynamic-login block-device-efi dhcp-all-interfaces devuser

镜像验证#

  • 启动虚拟机(x86_64)
# qemu-kvm
qemu-kvm \
  -enable-kvm \
  -m 4096 \
  -cpu host \
  -kernel image.kernel \
  -initrd image.initramfs \
  -append "root=/dev/ram0 console=ttyS0" \
  -nographic

# libvirt
virt-install \
  --name ipa-test \
  --memory 4096 \
  --vcpus 1 \
  --cpu host \
  --disk none \
  --boot kernel=/path/to/image.kernel,initrd=/path/to/image.initramfs,kernel_args="root=/dev/ram0 console=ttyS0" \
  --graphics none \
  --console pty,target_type=serial
